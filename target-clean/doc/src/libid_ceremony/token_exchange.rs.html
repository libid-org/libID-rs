1//! The GitHub Token Service contract of platform-ceremonies section 6.3.
2//!
3//! GitHub uses a confidential client, so the exchange cannot run in the
4//! browser: the client secret would have to go there. The deployment runs it
5//! instead, inside a notarized TLS session, and returns the attestation. The
6//! secret stays behind a range commitment and never reaches the browser.
7//!
8//! The service is stateless by requirement, not by preference. It holds
9//! ceremony credentials, so retention would create a compromise target with no
10//! protocol purpose (REQ-PLAT-42).
11//!
12//! # What this module is, and is not
13//!
14//! Section 6.3 names protocol values, not serialized field names: "the browser
15//! and deployment specifications own endpoint naming, transport framing,
16//! serialization, parsing bounds, caller authentication, and cache policy".
17//! So the route does not live here -- the deployment picks it, and the
18//! implementation that serves it states it.
19//!
20//! What lives here is the record pair and the bounds a served request and
21//! response must satisfy before the service acts on either. The semantics come
22//! from REQ-PLAT-37, -38, -41, -54 and -55; the byte bounds come from the
23//! GitHub token endpoint of the ceremony server contract, which is the
24//! deployment specification that owns them.
25
26pub const MAX_CODE_BYTES: usize = 1024;
27pub const CODE_VERIFIER_LEN: usize = 43;
28pub const MAX_ACCESS_TOKEN_BYTES: usize = 4096;
29/// The bearer commitment's blinder is fixed-width prover material, not a
30/// bounded string: the circuit opens exactly this many bytes.
31pub const BEARER_OPENING_LEN: usize = 16;
32pub const MAX_ATTESTED_DATA_BYTES: usize = 2 * 1024 * 1024;
33/// A recoverable secp256k1 signature: `r || s || v`.
34pub const SIGNATURE_LEN: usize = 65;
35pub const MAX_RESPONSE_BYTES: usize = 3 * 1024 * 1024;
36
37#[derive(Debug, thiserror::Error, PartialEq, Eq)]
38pub enum TokenExchangeError {
39    #[error("code is empty")]
40    EmptyCode,
41    #[error("code is {0} bytes, over the {MAX_CODE_BYTES}-byte bound")]
42    CodeTooLong(usize),
43    #[error("code carries a byte outside printable ASCII at index {0}")]
44    CodeNotPrintable(usize),
45    #[error("codeVerifier must match [A-Za-z0-9_-]{{43}}")]
46    MalformedCodeVerifier,
47    #[error("accessToken is empty")]
48    EmptyAccessToken,
49    #[error("accessToken is {0} bytes, over the {MAX_ACCESS_TOKEN_BYTES}-byte bound")]
50    AccessTokenTooLong(usize),
51    #[error("accessToken carries a byte outside printable ASCII at index {0}")]
52    AccessTokenNotPrintable(usize),
53    #[error("attestedData is empty")]
54    EmptyAttestedData,
55    #[error("attestedData is {0} bytes, over the {MAX_ATTESTED_DATA_BYTES}-byte bound")]
56    AttestedDataTooLong(usize),
57    #[error("signature is {0} bytes, not the {SIGNATURE_LEN} a notary signature is")]
58    SignatureWrongLength(usize),
59    #[error(
60        "bearerOpening is {0} bytes, not the {BEARER_OPENING_LEN} the circuit opens"
61    )]
62    BearerOpeningWrongLength(usize),
63}
64
65/// The index of the first byte outside printable ASCII, which excludes
66/// whitespace and control characters. Both credentials carried here are held to
67/// it: the code because it is echoed into a platform request, the bearer
68/// because it is echoed into an `Authorization` header.
69fn first_unprintable(s: &str) -> Option<usize> {
70    s.bytes().position(|b| !(0x21..=0x7e).contains(&b))
71}
72
73/// What the Canonical Runtime sends. Nothing else: the service uses only its
74/// compiled client identifier, secret, redirect URI, token endpoint and notary
75/// configuration, and accepts no caller-selected action, client, redirect,
76/// endpoint or return URL (REQ-PLAT-41).
77#[derive(Clone, Debug, PartialEq, Eq)]
78pub struct TokenRequest {
79    pub code: String,
80    pub code_verifier: String,
81}
82
83/// The signed attestation of the notarized exchange.
84///
85/// The bytes alone are not the attestation. REQ-PLAT-38 has the service return
86/// the attestation, and an attestation is a byte string together with the
87/// notary signature over it -- a record carrying only the bytes leaves the
88/// browser holding something no verifier can check, and no field to put the
89/// signature in.
90#[derive(Clone, Debug, PartialEq, Eq)]
91pub struct TokenAttestation {
92    /// The byte-exact attested data of the notarized exchange, preserved as
93    /// the notary produced it.
94    pub attested_data: Vec<u8>,
95    /// The notary signature authenticating those exact bytes.
96    pub signature: Vec<u8>,
97}
98
99/// What comes back. `access_token` and `bearer_opening` both stay inside the
100/// browser: the opening is private witness material for the Proving Circuit,
101/// and publishing it beside the commitment would publish the credential the
102/// commitment exists to hide (REQ-PLAT-55).
103#[derive(Clone, Debug, PartialEq, Eq)]
104pub struct TokenResponse {
105    pub access_token: String,
106    pub token_attestation: TokenAttestation,
107    /// The blinder that opens the committed bearer range of that attestation.
108    ///
109    /// Without it the browser holds the attestation and the bearer but cannot
110    /// build the proof: the blinder is prover-private material generated inside
111    /// a session only this service ran (REQ-PLAT-54).
112    pub bearer_opening: Vec<u8>,
113}
114
115impl TokenRequest {
116    /// Bounded parsing, per REQ-PLAT-37 and the request bounds of the server
117    /// contract.
118    pub fn validate(&self) -> Result<(), TokenExchangeError> {
119        if self.code.is_empty() {
120            return Err(TokenExchangeError::EmptyCode);
121        }
122        if self.code.len() > MAX_CODE_BYTES {
123            return Err(TokenExchangeError::CodeTooLong(self.code.len()));
124        }
125        if let Some(i) = first_unprintable(&self.code) {
126            return Err(TokenExchangeError::CodeNotPrintable(i));
127        }
128        if self.code_verifier.len() != CODE_VERIFIER_LEN
129            || !self
130                .code_verifier
131                .bytes()
132                .all(|b| b.is_ascii_alphanumeric() || b == b'_' || b == b'-')
133        {
134            return Err(TokenExchangeError::MalformedCodeVerifier);
135        }
136        Ok(())
137    }
138}
139
140impl TokenResponse {
141    /// Bounded parsing, per REQ-PLAT-38 and the response bounds of the server
142    /// contract.
143    ///
144    /// The three values are one result and the bounds say so: a bearer the
145    /// header cannot carry, an opening the circuit cannot use, or a signature
146    /// no recovery accepts each make the other two worthless, so each is exact
147    /// rather than merely capped.
148    pub fn validate(&self) -> Result<(), TokenExchangeError> {
149        if self.access_token.is_empty() {
150            return Err(TokenExchangeError::EmptyAccessToken);
151        }
152        if self.access_token.len() > MAX_ACCESS_TOKEN_BYTES {
153            return Err(TokenExchangeError::AccessTokenTooLong(
154                self.access_token.len(),
155            ));
156        }
157        if let Some(i) = first_unprintable(&self.access_token) {
158            return Err(TokenExchangeError::AccessTokenNotPrintable(i));
159        }
160        if self.token_attestation.attested_data.is_empty() {
161            return Err(TokenExchangeError::EmptyAttestedData);
162        }
163        if self.token_attestation.attested_data.len() > MAX_ATTESTED_DATA_BYTES {
164            return Err(TokenExchangeError::AttestedDataTooLong(
165                self.token_attestation.attested_data.len(),
166            ));
167        }
168        if self.token_attestation.signature.len() != SIGNATURE_LEN {
169            return Err(TokenExchangeError::SignatureWrongLength(
170                self.token_attestation.signature.len(),
171            ));
172        }
173        if self.bearer_opening.len() != BEARER_OPENING_LEN {
174            return Err(TokenExchangeError::BearerOpeningWrongLength(
175                self.bearer_opening.len(),
176            ));
177        }
178        Ok(())
179    }
180}

182#[cfg(test)]
183mod tests {
184    use super::*;
185
186    fn request() -> TokenRequest {
187        TokenRequest {
188            code: "abc123".into(),
189            code_verifier: "iMSTNh6gQkRnBGlY1c0MUOsD7MCO4G8C7ph1_gIZs5I".into(),
190        }
191    }
192
193    fn response() -> TokenResponse {
194        TokenResponse {
195            access_token: "gho_abc123".into(),
196            token_attestation: TokenAttestation {
197                attested_data: vec![0; 10],
198                signature: vec![0; SIGNATURE_LEN],
199            },
200            bearer_opening: vec![0; BEARER_OPENING_LEN],
201        }
202    }
203
204    #[test]
205    fn accepts_a_well_formed_request() {
206        request().validate().unwrap();
207    }
208
209    #[test]
210    fn accepts_a_well_formed_response() {
211        response().validate().unwrap();
212    }
213
214    #[test]
215    fn the_published_verifier_is_the_right_shape() {
216        // The section 7 conformance vector must satisfy the request bounds, or
217        // the service would refuse a verifier the specification itself
218        // produces.
219        assert_eq!(request().code_verifier.len(), CODE_VERIFIER_LEN);
220        request().validate().unwrap();
221    }
222
223    #[test]
224    fn refuses_an_empty_or_over_long_code() {
225        let mut r = request();
226        r.code = String::new();
227        assert_eq!(r.validate(), Err(TokenExchangeError::EmptyCode));
228        r.code = "a".repeat(MAX_CODE_BYTES + 1);
229        assert_eq!(
230            r.validate(),
231            Err(TokenExchangeError::CodeTooLong(MAX_CODE_BYTES + 1))
232        );
233    }
234
235    #[test]
236    fn refuses_whitespace_and_control_bytes_in_a_code() {
237        for bad in ["ab cd", "ab\tcd", "ab\ncd", "ab\0cd"] {
238            let r = TokenRequest {
239                code: bad.into(),
240                ..request()
241            };
242            assert!(
243                matches!(r.validate(), Err(TokenExchangeError::CodeNotPrintable(_))),
244                "accepted {bad:?}"
245            );
246        }
247    }
248
249    #[test]
250    fn refuses_a_verifier_of_the_wrong_length_or_charset() {
251        for bad in [
252            "short",
253            "iMSTNh6gQkRnBGlY1c0MUOsD7MCO4G8C7ph1_gIZs5", // 42
254            "iMSTNh6gQkRnBGlY1c0MUOsD7MCO4G8C7ph1_gIZs5II", // 44
255            "iMSTNh6gQkRnBGlY1c0MUOsD7MCO4G8C7ph1+gIZs5I", // base64, not base64url
256            "iMSTNh6gQkRnBGlY1c0MUOsD7MCO4G8C7ph1/gIZs5I",
257        ] {
258            let r = TokenRequest {
259                code_verifier: bad.into(),
260                ..request()
261            };
262            assert_eq!(
263                r.validate(),
264                Err(TokenExchangeError::MalformedCodeVerifier),
265                "accepted {bad:?}"
266            );
267        }
268    }

303    #[test]

305        let mut r = response();

317    #[test]
318    fn refuses_a_signature_that_is_not_exactly_recoverable_length() {
319        for len in [0, SIGNATURE_LEN - 1, SIGNATURE_LEN + 1] {
320            let mut r = response();
321            r.token_attestation.signature = vec![0; len];
322            assert_eq!(

330    #[test]
331    fn refuses_an_opening_that_is_not_exactly_what_the_circuit_opens() {
332        // A near miss is the dangerous one: a bounded check accepted both of
333        // these, and the circuit accepts neither.
334        for len in [0, BEARER_OPENING_LEN - 1, BEARER_OPENING_LEN + 1, 256] {
335            let mut r = response();
336            r.bearer_opening = vec![0; len];

339                Err(TokenExchangeError::BearerOpeningWrongLength(len)),

342        }
