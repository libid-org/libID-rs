1//! The attested-data format the launch profiles pin.
2//!
3//! THE LAYOUT IS THE PROFILE'S, NOT THE SPECIFICATION'S. `REQ-COMMON-18` has a
4//! Platform Profile fix the attestation format it accepts and leaves the format
5//! itself to the profile author. So this module is the definition, not a
6//! reading of one, and every rule it keeps is stated here in full.
7//!
8//! An attestation is a byte string and a signature over it. The Notary Service
9//! signs off chain, where it holds the transcript, and verifies on chain, where
10//! it holds none. The verifying side therefore rebuilds these exact bytes from
11//! what it was handed and derives the signing key from them: a field reordered,
12//! omitted, or encoded differently on either side derives a key nobody trusts,
13//! and `REQ-COMMON-33` leaves it nothing else to check the signature against.
14//!
15//! Every boundary is derivable from bytes that precede it, so decoding is one
16//! forward pass and two different attestations cannot share one preimage by
17//! shifting a boundary.
18//!
19//! Four components must agree on these bytes: this crate, the Solidity
20//! decoder, the TypeScript mirror, and the notary that signs them. A
21//! divergence is silent -- the signature derives a key nobody trusts and every
22//! genuine attestation is rejected with no error saying why.
23
24use libid_crypto::keccak256;
25
26/// Offsets are zero-based into that direction's complete transcript, `start`
27/// inclusive and `end` exclusive.
28#[derive(Clone, Debug, PartialEq, Eq, bincode::Encode)]
29pub struct RevealedRange {
30    pub start: u32,
31    /// The range's plaintext. Its length IS the range's length -- there is no
32    /// `end`, because two ways to say the same thing is one way to disagree.
33    /// The decoder computes `end = start + bytes.len()`.
34    pub bytes: Vec<u8>,
35}
36
37/// A hidden range, carried as its offsets and a blinded commitment. The
38/// plaintext of a committed range never appears in the attested data.
39#[derive(Clone, Debug, PartialEq, Eq, bincode::Encode)]
40pub struct RangeCommitment {
41    pub start: u32,
42    pub end: u32,
43    pub commitment: [u8; 32],
44}
45
46/// One direction of the session.
47#[derive(Clone, Debug, Default, PartialEq, Eq, bincode::Encode)]
48pub struct DirectionBlock {
49    pub revealed: Vec<RevealedRange>,
50    pub commitments: Vec<RangeCommitment>,
51}
52
53/// The signed bytes.
54///
55/// The attested data describes the observed session and says nothing about
56/// where the evidence will be spent: no chain, no verifier identity. The
57/// Authorization Digest already commits the chain, and binding an attestation
58/// to one verifier would stop a newly registered version checking attestations
59/// made before it existed.
60#[derive(Clone, Debug, PartialEq, Eq, bincode::Encode)]
61pub struct AttestedData {
62    /// The TLS server name the notary authenticated, hashed.
63    ///
64    /// This is the only identity in the record, and the notary observed it
65    /// rather than being told it. Which platform that host belongs to, and
66    /// which session of a ceremony this is, are read from the revealed request
67    /// line by the party that pins those constants.
68    pub authority_id: [u8; 32],
69    pub created_at: u64,
70    pub sent_transcript_length: u32,
71    pub recv_transcript_length: u32,
72    pub sent: DirectionBlock,
73    pub received: DirectionBlock,
74}
75
76/// Bytes before the first direction block: the authority, `createdAt`, and the
77/// two transcript lengths.
78pub const HEADER_LEN: usize = 32 + 8 + 4 + 4;
79
80/// Big-endian, fixed-width, no varints: the decoder is Solidity, which has no
81/// use for a compact integer that costs a branch to read.
82///
83/// Pinned to one exact bincode version in `Cargo.toml`. These bytes are a
84/// signed preimage, so a layout change in a patch release would silently
85/// change what every notary signs -- and the cross-language fixture below is
86/// what would catch it.
87const WIRE: bincode::config::Configuration<
88    bincode::config::BigEndian,
89    bincode::config::Fixint,
90> = bincode::config::standard()
91    .with_big_endian()
92    .with_fixed_int_encoding();
93
94impl AttestedData {
95    /// The `authority_id` of a record covering a session with `server_name`:
96    /// keccak256 over the canonical authority bytes (REQ-COMMON-21,
97    /// REQ-COMMON-21A).
98    ///
99    /// An associated function on the record rather than a free `tag`, because
100    /// the free form said nothing about what may be hashed. The one input this
101    /// field accepts is the TLS server name the notary AUTHENTICATED. A `Host`
102    /// header the prover composed hashes just as well and yields a record
103    /// naming an authority nobody observed -- and that substitution is one the
104    /// transcript cannot rule out, since the request carries the authority only
105    /// where the prover wrote it. Naming the record puts the rule beside the
106    /// field.
107    ///
108    /// Canonicalization is ASCII lowercase, and it happens HERE rather than at
109    /// each caller. The id is compared on chain against a constant a profile
110    /// pins, and ASCII case is the one difference a TLS stack hands back
111    /// without anyone noticing: `API.x.com` authenticates the same server and
112    /// hashes to a different id. Left to the call site it is a step every
113    /// future caller has to remember, and the one that forgets produces
114    /// attestations that are signed, well formed, and refused by every verifier
115    /// with nothing pointing at the capital letter. It sat at the one caller in
116    /// `libid-tlsn` while the other passed a string that was already lowercase
117    /// -- a rule kept by accident.
118    ///
119    /// Section 9 also asks for no trailing dot, and this does NOT strip one,
120    /// exactly as `tag` did not. Changing what a signed field hashes belongs in
121    /// a change that argues for it and tests it, not in a rename.
122    ///
123    /// A string rather than a server-name type: this crate is published and
124    /// knows nothing about how a TLS library models a name, which is also what
125    /// keeps this mapping testable without a session.
126    ///
127    /// The record's one remaining 32-byte tag. It used to serve three more --
128    /// format, platform and session -- and those went with the fields the
129    /// notary was handed rather than saw.
130    pub fn authority_id_of(server_name: &str) -> [u8; 32] {
131        keccak256(server_name.to_ascii_lowercase().as_bytes())
132    }
133
134    /// Lay the record out. This does NOT judge it: a malformed record is the
135    /// prover's problem, the Platform Verifier's decision, and the client's to
136    /// catch in a dry run. Refusing to sign here would only withhold a session
137    /// the notary really did observe.
138    ///
139    /// The layout is the struct above, in declaration order. Nothing here
140    /// restates it, so nothing here can drift from it.
141    pub fn encode(&self) -> Result<Vec<u8>, bincode::error::EncodeError> {
142        bincode::encode_to_vec(self, WIRE)
143    }
144
145    /// What the notary signs, and the only preimage it ever signs
146    /// (REQ-COMMON-33).
147    pub fn digest(&self) -> Result<[u8; 32], bincode::error::EncodeError> {
148        Ok(keccak256(&self.encode()?))
149    }
150}

156    fn sample() -> AttestedData {
157        // Shaped like the X identity session: the request reveals everything
158        // but the bearer, which is committed and framed by the header bytes.
159        AttestedData {
160            authority_id: AttestedData::authority_id_of("api.x.com"),
161            created_at: 1_770_000_000,
162            sent_transcript_length: 60,
163            recv_transcript_length: 40,
164            sent: DirectionBlock {
165                revealed: vec![
166                    RevealedRange {
167                        start: 0,
168                        bytes: vec![b'a'; 20],
169                    },
170                    RevealedRange {
171                        start: 40,
172                        bytes: vec![b'b'; 20],
173                    },
174                ],
175                commitments: vec![RangeCommitment {
176                    start: 20,
177                    end: 40,
178                    commitment: [7u8; 32],
179                }],
180            },
181            received: DirectionBlock {
182                revealed: vec![RevealedRange {
183                    start: 0,
184                    bytes: vec![b'c'; 10],
185                }],
186                commitments: vec![RangeCommitment {
187                    start: 10,
188                    end: 40,
189                    commitment: [9u8; 32],
190                }],
191            },
192        }
193    }

195    /// The exact bytes `solidity/contracts/ceremony/test/CeremonyAttestation.t.sol`
196    /// decodes. Both sides carry this fixture, so a change to either encoder

200
201    const CROSS_LANGUAGE_DIGEST: &str =
202        "48162f05bdb27b19b3544bf2aae608745861bf357bb31e07f536b6fb50e95936";

204    #[test]
205    fn agrees_with_the_solidity_decoder() {
206        let encoded = sample().encode().unwrap();
207        assert_eq!(hex::encode(&encoded), CROSS_LANGUAGE_FIXTURE);
208        assert_eq!(
209            hex::encode(sample().digest().unwrap()),
210            CROSS_LANGUAGE_DIGEST
211        );
212    }
213
214    #[test]
215    fn the_authority_is_hashed_in_one_canonical_spelling() {
216        // REQ-COMMON-21A fixes the preimage as the lowercase ASCII server name.
217        // A profile pins this id as a constant, so a differently cased spelling
218        // of the same authenticated host would hash to an id no profile matches
219        // -- every genuine attestation for that host refused, with nothing
220        // pointing at the capital letter.
221        assert_eq!(
222            AttestedData::authority_id_of("API.X.com"),
223            AttestedData::authority_id_of("api.x.com")
224        );
225    }
226
227    #[test]
228    fn header_is_one_hundred_and_forty_four_bytes() {
229        let mut data = sample();
230        data.sent = DirectionBlock::default();
231        data.received = DirectionBlock::default();
232        data.sent_transcript_length = 0;
233        data.recv_transcript_length = 0;
234        // Header plus two empty counts per direction.
235        // Four counts, one per direction per list, eight bytes each.
236        assert_eq!(data.encode().unwrap().len(), HEADER_LEN + 4 * 8);
237        assert_eq!(HEADER_LEN, 48);
238    }
239
240    #[test]
241    fn every_header_field_changes_the_digest() {
242        let base = sample().digest().unwrap();
243        for mutate in [
244            (|d: &mut AttestedData| d.authority_id[0] ^= 1) as fn(&mut AttestedData),
245            |d| d.created_at += 1,
246        ] {
247            let mut data = sample();
248            mutate(&mut data);
249            assert_ne!(data.digest().unwrap(), base);
250        }
251    }
252
253    #[test]
254    fn two_sessions_of_one_ceremony_are_not_interchangeable() {
255        // Nothing in the record labels which session it covers, and nothing
256        // needs to: the sessions differ in what the notary OBSERVED. The
257        // request line is a revealed range, and the verifier compares it
258        // against the path its profile pins, so a token attestation offered in
259        // the identity slot fails on bytes the notary actually saw rather than
260        // on a label it was handed.
261        let mut token = sample();
262        token.sent.revealed[0].bytes = b"POST /2/oauth2/token ".to_vec();
263        assert_ne!(token.digest().unwrap(), sample().digest().unwrap());
264    }
265
266    #[test]
267    fn a_shifted_boundary_cannot_produce_one_preimage() {
268        // Moving a byte from a revealed range into the next one changes the
269        // encoding: the range's length is its bytes, and the following span
270        // starts one earlier.
271        let mut moved = sample();
272        moved.sent.revealed[0].bytes.pop();
273        moved.sent.commitments[0].start = 19;
274        assert_ne!(moved.digest().unwrap(), sample().digest().unwrap());
275    }
