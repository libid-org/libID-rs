1//! Turn what a notarized session produced into the attested data of
2//! ceremony-common section 9.1.
3//!
4//! This is the only place tlsn's view of a transcript meets libID's. The
5//! layering is deliberate: `libid-ceremony` owns the bytes and is publishable,
6//! this crate owns the translation and is git-only because tlsn is. Nothing
7//! above needs to know that a `RangeSet` exists.
8//!
9//! The byte layout itself is the profile's rather than the specification's;
10//! `libid_ceremony::attestation` says under which requirement, and states each
11//! rule it keeps in full.
12

28/// One notarized session, as the notary observed it.
29///
30/// Every field here is something the notary SAW: the transcript it helped
31/// decrypt, the server name it authenticated against WebPKI, the commitments
32/// the prover made inside the session, and the moment its own clock said the
33/// session closed. That is the line REQ-COMMON-33 draws between what a notary
34/// may sign and what it may not, and a type is how the line is kept -- a value
35/// the notary was merely TOLD has no field to arrive in, so it cannot reach the
36/// signed bytes by being appended to an argument list.
37///
38/// It borrows rather than owns. The party that ran the session already holds
39/// every one of these, and copying a whole transcript across in order to
40/// describe it would double the peak memory of a notarization to say nothing
41/// new. `Copy` for the same reason: handing the same view to two calls should
42/// not mean restating it.
43///
44/// Deliberately not `Debug`. Printing one prints the revealed transcript --
45/// the prover's request with its credential framing around it -- into whatever
46/// log line was being written at the time.
47#[derive(Clone, Copy)]
48pub struct ObservedSession<'a> {
49    /// The transcript with the prover's revealed ranges opened and the rest
50    /// still closed. Both directions and both signed lengths are read from this
51    /// one value, so there is no pair of lengths that can disagree with the
52    /// ranges they bound.
53    pub transcript: &'a PartialTranscript,
54    /// The DNS name the notary authenticated, which the caller takes from
55    /// `ServerName::Dns`.
56    ///
57    /// It arrives as a string rather than as tlsn's name type so this mapping
58    /// stays testable and so nothing here depends on how upstream models a
59    /// server name. It reaches the record as a signed field rather than as a
60    /// transcript range because the transcript carries the authority only in a
61    /// prover-composed `Host` header, which says nothing about which server
62    /// answered (REQ-COMMON-21, REQ-COMMON-21A).
63    pub authority: &'a str,
64    /// Every commitment the session produced, both directions together, in
65    /// whatever order the prover made them. [`ObservedSession::attested_data`]
66    /// splits them by direction and sorts them by offset, so a caller passes on
67    /// what it was handed rather than pre-sorting a list the format reorders
68    /// anyway.
69    pub commitments: &'a [TranscriptCommitment],
70    /// The notary's OWN clock reading when the session completed.
71    ///
72    /// Never the prover's, never a response header, never any other party's:
73    /// the verifier's freshness window is measured from this, so a reading the
74    /// observed party could choose would be a window it could choose. It is a
75    /// field rather than a call to the clock here so a test can pin it; the
76    /// caller must pass its own.
77    pub created_at: u64,
78}

80#[derive(Debug, thiserror::Error)]
81pub enum AttestError {
82    #[error(
83        "a commitment uses {0:?}, but REQ-COMMON-38 pins SHA-256 for launch profiles"
84    )]
85    WrongCommitmentAlgorithm(HashAlgId),
86    #[error("a commitment covers {0} disjoint ranges; the format carries one range per commitment")]
87    DisjointCommitment(usize),
88    #[error("a commitment hash is {0} bytes, not 32")]
89    BadCommitmentLength(usize),
90    #[error("transcript offset {0} does not fit the format's 32-bit field")]
91    OffsetTooLarge(usize),
92}
93
94fn u32_of(value: usize) -> Result<u32, AttestError> {
95    u32::try_from(value).map_err(|_| AttestError::OffsetTooLarge(value))
96}

98impl ObservedSession<'_> {
99    /// Lay this session out as the [`AttestedData`] of ceremony-common
100    /// section 9.1.
101    ///
102    /// A method rather than a constructor because the constructor form is not
103    /// available here: `AttestedData` is defined in `libid-ceremony`, which is
104    /// published to crates.io and must never name a tlsn type, so an inherent
105    /// impl for it cannot live in this crate. The object that owns the inputs
106    /// carries the producer instead -- and the four arguments this replaces
107    /// were never four unrelated things. They were four readings of one
108    /// session, which a caller had to keep in step by hand.
109    ///
110    /// The notary places nothing here that it derived by applying a profile
111    /// rule -- no handle, no account identifier, no client identifier, no chain
112    /// address (REQ-COMMON-33). Every such value is already derivable from the
113    /// revealed ranges, a second signed copy can disagree with the bytes it
114    /// came from, and producing one would make the Notary Service decide
115    /// something profile-specific. What is signed is what [`ObservedSession`]
116    /// holds, in the order the record declares it.
117    ///
118    /// This fails only where the session cannot be described by the format at
119    /// all: an offset past its 32-bit field, a commitment under the wrong hash,
120    /// a commitment over disjoint ranges. It judges nothing else. Whether the
121    /// ranges tile, whether the request carries exactly one credential header
122    /// -- those are the Platform Verifier's decision and the client's dry run,
123    /// and refusing here would only withhold a session the notary really did
124    /// observe.
125    pub fn attested_data(&self) -> Result<AttestedData, AttestError> {
126        Ok(AttestedData {
127            authority_id: AttestedData::authority_id_of(self.authority),
128            created_at: self.created_at,
129            sent_transcript_length: u32_of(self.transcript.len_sent())?,
130            recv_transcript_length: u32_of(self.transcript.len_received())?,
131            sent: self.direction_block(Direction::Sent)?,
132            received: self.direction_block(Direction::Received)?,
133        })
134    }
135
136    /// One direction's revealed runs and its commitments, both in ascending
137    /// start order.
138    ///
139    /// The direction is the only parameter, because everything else it reads
140    /// belongs to the session -- and the two it used to take apart, the
141    /// transcript and the commitment list, must come from the SAME session or
142    /// the block describes bytes nobody observed together. Holding them on
143    /// `self` makes that pairing unspellable-wrong rather than merely
144    /// uncommon.
145    ///
146    /// Private, and not a constructor: `DirectionBlock` is `libid-ceremony`'s,
147    /// so a constructor for it cannot live here either, and nothing outside
148    /// this file has a reason to build one direction alone.
149    fn direction_block(
150        &self,
151        direction: Direction,
152    ) -> Result<DirectionBlock, AttestError> {
153        let (authed, data) = match direction {
154            Direction::Sent => {
155                (self.transcript.sent_authed(), self.transcript.sent_unsafe())
156            }
157            Direction::Received => (
158                self.transcript.received_authed(),
159                self.transcript.received_unsafe(),
160            ),
161        };
162
163        // One entry per revealed range, in ascending start order, each carrying
164        // where it sat and what it held. Revealed bytes signed without their
165        // offsets say that some bytes were disclosed but not where they sat, which
166        // is not enough to tile a transcript. The end is the bytes' own length, so
167        // it is not written down twice.
168        let mut revealed = Vec::new();
169        for range in authed.iter() {
170            // Still checked, even though only `start` is encoded: a range whose end
171            // does not fit is a transcript this record cannot describe.
172            u32_of(range.end)?;
173            revealed.push(RevealedRange {
174                start: u32_of(range.start)?,
175                bytes: data[range.clone()].to_vec(),
176            });
177        }
178
179        let mut out = Vec::new();
180        for commitment in self.commitments {
181            // The enum is non-exhaustive upstream, so an unknown commitment kind
182            // is skipped rather than assumed to be a hash.
183            let TranscriptCommitment::Hash(hash) = commitment else {
184                continue;
185            };
186            if hash.direction != direction {
187                continue;
188            }
189            // The notarization library defaults to BLAKE3 while the Proving Circuit
190            // computes SHA-256, so a prover left on library defaults produces
191            // commitments the circuit cannot open (REQ-COMMON-38).
192            if hash.hash.alg != HashAlgId::SHA256 {
193                return Err(AttestError::WrongCommitmentAlgorithm(hash.hash.alg));
194            }
195
196            // A `RangeSet` may be disjoint, but the format pairs one commitment
197            // value with one offset pair. A hash over a union cannot be split
198            // between two entries without inventing a value for each.
199            let ranges: Vec<_> = hash.idx.iter().collect();
200            let [range] = ranges.as_slice() else {
201                return Err(AttestError::DisjointCommitment(ranges.len()));
202            };
203
204            let value = hash.hash.value.as_bytes();
205            let value: [u8; 32] = value
206                .try_into()
207                .map_err(|_| AttestError::BadCommitmentLength(value.len()))?;
208
209            out.push(RangeCommitment {
210                start: u32_of(range.start)?,
211                end: u32_of(range.end)?,
212                commitment: value,
213            });
214        }
215        out.sort_by_key(|c| c.start);
216
217        Ok(DirectionBlock {
218            revealed,
219            commitments: out,
220        })
221    }
222}

226    /// The Platform Verifier requires the revealed ranges and the commitments
227    /// to account for the signed length exactly. That is its rule to enforce,
228    /// not ours -- but a layout that cannot satisfy it produces attestations no
229    /// verifier accepts, so it is worth asserting here on the way out.
230    fn assert_tiles(block: &libid_ceremony::DirectionBlock, length: u32) {
231        let mut spans: Vec<(u32, u32)> = block
232            .revealed
233            .iter()
234            .map(|r| (r.start, r.start + r.bytes.len() as u32))
235            .chain(block.commitments.iter().map(|c| (c.start, c.end)))
236            .collect();
237        spans.sort_unstable();
238        let mut at = 0u32;
239        for (start, end) in spans {
240            assert_eq!(start, at, "gap or overlap before {start}");
241            at = end;
242        }
243        assert_eq!(at, length, "the spans do not reach the signed length");
244    }

287    /// `Hash` has no public constructor, so build it the way upstream
288    /// deserializes it: a sequence of bytes.
289    fn hash32(byte: u8) -> TypedHash {
290        TypedHash {
291            alg: HashAlgId::SHA256,
292            value: serde_json::from_value(serde_json::json!(vec![byte; 32])).unwrap(),
293        }
294    }

296    /// Reveal everything except the bearer, and commit the bearer -- the shape
297    /// an identity session actually produces.
298    fn session() -> (PartialTranscript, Vec<TranscriptCommitment>) {
299        let bearer = 45..48; // "TOK"
300        let transcript = Transcript::new(SENT, RECV);
301        let sent_revealed = RangeSet::from(vec![0..bearer.start, bearer.end..SENT.len()]);
302        let partial = transcript.to_partial(sent_revealed, RangeSet::from(0..RECV.len()));
303        let commitments = vec![TranscriptCommitment::Hash(PlaintextHash {
304            direction: Direction::Sent,
305            idx: RangeSet::from(bearer),
306            hash: hash32(7),
307        })];
308        (partial, commitments)
309    }

321    #[test]
322    fn encodes_to_the_length_its_own_fields_imply() {
323        let (partial, commitments) = session();
324        let data = observed(&partial, &commitments).attested_data().unwrap();
325        let encoded = data.encode().unwrap();
326
327        // No decoder here to round-trip against: decoding is the chain's and
328        // the client's. What stays checkable on this side is that every byte
329        // the fields describe is present, which is the property the layout
330        // gives the forward-parsing decoder something to walk.
331        let mut want = libid_ceremony::attestation::HEADER_LEN;
332        for d in [&data.sent, &data.received] {
333            want += 8 + 8; // one eight-byte count per list
334            for r in &d.revealed {
335                want += 4 + 8 + r.bytes.len(); // start, byte length, bytes
336            }
337            want += d.commitments.len() * (4 + 4 + 32); // start, end, commitment
338        }
339        assert_eq!(encoded.len(), want);
340    }

412    #[test]
413    fn places_no_profile_derived_value_in_the_signed_bytes() {
414        // REQ-COMMON-33: the Notary Service decides nothing profile-specific,
415        // so the notary places no value it obtained by applying a profile rule
416        // -- no handle, no account identifier, no client identifier, no chain
417        // address. Every one is already derivable
418        // from the revealed ranges, and a second signed representation can
419        // disagree with the bytes it was taken from.
420        //
421        // Tested structurally: two sessions whose responses name different
422        // accounts must produce IDENTICAL header bytes. If any identity field
423        // were signed into the header, it would differ here.
424        let other_recv: &[u8] = b"HTTP/1.1 200 OK\r\n\r\n{\"id\":\"9\"}";
425        assert_eq!(
426            other_recv.len(),
427            RECV.len(),
428            "the two responses must be the same length"
429        );
430
431        let bearer = 45..48;
432        let sent_revealed = RangeSet::from(vec![0..bearer.start, bearer.end..SENT.len()]);
433        let commitments = vec![TranscriptCommitment::Hash(PlaintextHash {
434            direction: Direction::Sent,
435            idx: RangeSet::from(bearer),
436            hash: hash32(7),
437        })];
438
439        let mut headers = Vec::new();
440        for recv in [RECV, other_recv] {
441            let partial = Transcript::new(SENT, recv)
442                .to_partial(sent_revealed.clone(), RangeSet::from(0..recv.len()));
443            let data = observed(&partial, &commitments).attested_data().unwrap();
444            headers.push(data.encode().unwrap()[..144].to_vec());
445        }
446        assert_eq!(
447            headers[0], headers[1],
448            "an identity field leaked into the signed header"
449        );
450
451        // And the accounts really are different, so the test is not vacuous.
452        let a = Transcript::new(SENT, RECV)
453            .to_partial(sent_revealed.clone(), RangeSet::from(0..RECV.len()));
454        let b = Transcript::new(SENT, other_recv)
455            .to_partial(sent_revealed, RangeSet::from(0..other_recv.len()));
456        assert_ne!(
457            observed(&a, &commitments)
458                .attested_data()
459                .unwrap()
460                .encode()
461                .unwrap(),
462            observed(&b, &commitments)
463                .attested_data()
464                .unwrap()
465                .encode()
466                .unwrap(),
467            "the two sessions must differ somewhere -- in the revealed range"
468        );
469    }

471    // --- The layouts the prover selects must satisfy the verifier ----------
472
473    /// Build a session from a real transcript plus the layout the ceremony
474    /// selects for it, and check the attested data it produces TILES.
475    ///
476    /// This is the property no unit test on either side reaches on its own. The
477    /// verifier demands exact coverage; the prover chooses the ranges. If they
478    /// disagree, every check passes in isolation and no honest ceremony
479    /// verifies -- a liveness failure that only shows up in an end-to-end run.
480    fn round_trip(
481        sent: &[u8],
482        recv: &[u8],
483        sent_layout: &Layout,
484        recv_layout: &Layout,
485    ) -> AttestedData {
486        let transcript = Transcript::new(sent, recv);
487        let partial = transcript.to_partial(
488            RangeSet::from(sent_layout.reveal.clone()),
489            RangeSet::from(recv_layout.reveal.clone()),
490        );
491        let mut commitments = Vec::new();
492        for (direction, l) in [
493            (Direction::Sent, sent_layout),
494            (Direction::Received, recv_layout),
495        ] {
496            for range in &l.commit {
497                commitments.push(TranscriptCommitment::Hash(PlaintextHash {
498                    direction,
499                    idx: RangeSet::from(range.clone()),
500                    hash: hash32(1),
501                }));
502            }
503        }
504        observed(&partial, &commitments).attested_data().unwrap()
505    }

507    #[test]
508    fn the_identity_session_layout_tiles_both_directions() {
509        let sent: &[u8] = b"GET /2/users/me HTTP/1.1\r\nhost: api.x.com\r\nauthorization: Bearer TOKENVALUE\r\nconnection: close\r\n\r\n";
510        let recv: &[u8] = b"HTTP/1.1 200 OK\r\ncontent-type: application/json\r\n\r\n{\"data\":{\"id\":\"2244994945\",\"name\":\"Al\",\"username\":\"alice\"}}";
511
512        let s = Layout::identity_request(sent).unwrap();
513        let r = Layout::identity_response(recv, "id", IdShape::JsonString, "username")
514            .unwrap();
515        let data = round_trip(sent, recv, &s, &r);
516        assert_tiles(&data.sent, data.sent_transcript_length);
517        assert_tiles(&data.received, data.recv_transcript_length);
518
519        // And exactly one credential is hidden in the request, which is what
520        // ties the framed range to the one the circuit opens.
521        assert_eq!(data.sent.commitments.len(), 1);
522    }
