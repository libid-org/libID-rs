1//! Choosing what a notarized session reveals.
2//!
3//! The Platform Verifier checks that the revealed ranges and the commitments
4//! TILE the transcript: every byte accounted for, no gap and no overlap. A gap
5//! is where a prover hides bytes, so a session that leaves one is refused --
6//! which means the selection here is not a disclosure preference, it is a
7//! correctness requirement. Choose the wrong ranges and no honest ceremony
8//! verifies at all.
9//!
10//! Every layout below therefore names only what it REVEALS, and the commitments
11//! are derived as the complement. Tiling then holds by construction rather than
12//! by inspection.
13//!
14//! Nothing here is applied on anyone's behalf. A prover notarizing a ceremony
15//! session calls these and hands the result to `prover_generic`; a prover doing
16//! something else states its own. In Rust that prover will be the GitHub
17//! Token-Exchange Service, for the token session. The other three sessions are
18//! the browser's.
19

28/// What one direction of one session discloses.
29#[derive(Clone, Debug, PartialEq, Eq)]
30pub struct Layout {
31    /// Ascending, non-overlapping.
32    pub reveal: Vec<Range<usize>>,
33    /// The complement of `reveal` over the whole direction.
34    pub commit: Vec<Range<usize>>,
35}
36
37#[derive(Debug, thiserror::Error, PartialEq, Eq)]
38pub enum LayoutError {
39    #[error("the request has no `{0}` header, so the layout has nothing to anchor on")]
40    MissingHeader(&'static str),
41    #[error("the response carries no `{0}` field where the profile expects one")]
42    MissingField(String),
43    #[error("the transcript has no head boundary, so its body cannot be located")]
44    NoHeadBoundary,
45    #[error("the credential to commit was not found in the request body")]
46    MissingCredential,
47}
48
49/// The bytes of `[0, len)` that `reveal` does not cover.
50///
51/// Deriving the commitments this way is what makes every layout tile. The
52/// alternative -- listing both and hoping they agree -- is the mistake the
53/// verifier exists to catch.
54fn complement(reveal: &[Range<usize>], len: usize) -> Vec<Range<usize>> {
55    let mut out = Vec::new();
56    let mut at = 0usize;
57    for r in reveal {
58        if r.start > at {
59            out.push(at..r.start);
60        }
61        at = r.end;
62    }
63    if at < len {
64        out.push(at..len);
65    }
66    out
67}

69/// Which shape the platform's immutable identifier takes.
70#[derive(Clone, Copy, Debug, PartialEq, Eq)]
71pub enum IdShape {
72    /// X: `"id":"2244994945"`.
73    JsonString,
74    /// GitHub: `"id":583231,` -- the terminator is revealed with it, because it
75    /// is what proves the digits are the whole number rather than a prefix.
76    JsonInteger,
77}
78
79impl Layout {
80    /// A layout that reveals these ranges of a direction `len` bytes long, and
81    /// commits everything they leave.
82    ///
83    /// The one door into a tiling `Layout`, and the reason every constructor below
84    /// tiles by construction rather than by inspection: no constructor states
85    /// `commit`, so no constructor's list can disagree with the reveals it was
86    /// supposed to complement. `layout` -- a lowercase homonym of the type it
87    /// built, in a module about layouts -- said none of that.
88    ///
89    /// `complement` walks the reveals once, taking each as starting where the last
90    /// one ended, so unsorted input reads as overlap and yields a complement that
91    /// tiles nothing -- which the Platform Verifier rejects and nothing here would
92    /// catch. Sorting is done once, here, so no caller has to remember: the
93    /// layouts that build in order are unaffected, and
94    /// [`Layout::identity_response`], whose two members arrive in whatever order
95    /// the platform serialized them, carries no sort of its own. Overlapping input
96    /// is a caller bug that sorting cannot repair, and the debug assertion is what
97    /// says so.
98    ///
99    /// Private, and `Layout`'s fields stay public beside it. Callers outside this
100    /// module state layouts this module does not know -- `libid-tlsn`'s legacy
101    /// prover builds two whose ranges deliberately do not tile -- so tiling is a
102    /// property of these constructors and not of the type. A public constructor
103    /// advertising a guarantee the type does not enforce would be worse than no
104    /// public constructor at all.
105    ///
106    /// It takes an iterator rather than a `Vec` so that a one-range layout is
107    /// spelled `core::iter::once(a..b)`. Both `vec![a..b]` and `[a..b]` trip
108    /// `clippy::single_range_in_vec_init`, a lint that exists to catch `vec![0; n]`
109    /// typos and cannot tell this apart from one; the iterator form says what is
110    /// meant without a named helper standing in for it.
111    fn revealing(reveal: impl IntoIterator<Item = Range<usize>>, len: usize) -> Self {
112        let mut reveal: Vec<Range<usize>> = reveal.into_iter().collect();
113        reveal.sort_by_key(|r| r.start);
114        debug_assert!(
115            reveal.windows(2).all(|pair| pair[0].end <= pair[1].start),
116            "reveal ranges overlap: {reveal:?}"
117        );
118        let commit = complement(&reveal, len);
119        Layout { reveal, commit }
120    }
121
122    /// The token request of `x/v1`, or the token exchange of `github/v1`.
123    ///
124    /// X reveals the request whole: it authenticates with a public client, so the
125    /// request carries nothing secret and the head boundary stays visible, which is
126    /// how the verifier locates the body at all. GitHub commits its `client_secret`
127    /// alone -- ordered last in the body, so the revealed run is a prefix and the
128    /// commitment reaches the transcript end.
129    pub fn token_request(
130        sent: &[u8],
131        secret_field: Option<&str>,
132    ) -> Result<Self, LayoutError> {
133        let Some(field) = secret_field else {
134            return Ok(Self::revealing(core::iter::once(0..sent.len()), sent.len()));
135        };
136
137        // `&client_secret=` begins the committed tail. The profile orders it last
138        // under REQ-COMMON-22 precisely so this is a suffix and not a hole.
139        let needle = format!("&{field}=");
140        let start = sent
141            .windows(needle.len())
142            .position(|w| w == needle.as_bytes())
143            .ok_or(LayoutError::MissingCredential)?;
144        Ok(Self::revealing(core::iter::once(0..start), sent.len()))
145    }
146
147    /// The token response: the `"access_token":"` delimiter and its closing quote
148    /// are revealed, and everything else -- the bearer included -- is committed.
149    ///
150    /// Those two anchors are what identify the committed bearer. Without them the
151    /// committed range is indistinguishable from a `refresh_token` value, or any
152    /// other substring the prover chose to commit (REQ-PLAT-57, REQ-PLAT-58).
153    pub fn token_response(recv: &[u8]) -> Result<Self, LayoutError> {
154        // Named once, and a constant rather than a parameter. `access_token` is
155        // RFC 6749 section 5.1, not a platform's choice -- which is why the
156        // contract pins `ACCESS_TOKEN_PREFIX` on `TlsNotaryVerifierBase`, shared by
157        // every profile, while the things that ARE platform choices are per-profile
158        // virtuals there and parameters here: the committed body credential of
159        // `Layout::token_request`, the field names of
160        // `Layout::identity_response`.
161        const FIELD: &str = "access_token";
162        let missing = || LayoutError::MissingField(FIELD.into());
163
164        // Through the shared reader rather than a scan of its own. That one locates
165        // the response BODY, so a header carrying this delimiter cannot answer
166        // first, and it refuses a member that chunk framing runs through -- which
167        // this direction cares about most, because the framing would land inside
168        // the committed bearer and the circuit would open a value the token service
169        // never returned.
170        let found = JsonMember::in_response(recv, FIELD).ok_or_else(missing)?;
171
172        // Reveal the two delimiters and let the complement commit the bearer
173        // between them. Both boundaries come from the scan that found the member,
174        // so nothing here restates `"access_token":"` to recompute one.
175        //
176        // An empty bearer is refused: it would leave the two reveals adjacent and
177        // commit nothing, and a response direction with no commitment is one the
178        // framing check on chain finds no bearer in.
179        if found.value.is_empty() {
180            return Err(missing());
181        }
182
183        Ok(Self::revealing(
184            [
185                found.member.start..found.value.start,
186                found.value.end..found.member.end,
187            ],
188            recv.len(),
189        ))
190    }
191
192    /// The identity request: every byte revealed except the bearer value, which is
193    /// committed.
194    ///
195    /// The two revealed runs plus the committed one account for the request exactly,
196    /// which is what REQ-COMMON-35 demands and what leaves the committed range as
197    /// the only region the verifier cannot read.
198    pub fn identity_request(sent: &[u8]) -> Result<Self, LayoutError> {
199        const PREFIX: &[u8] = b"\r\nauthorization: Bearer ";
200        let prefix_at = sent
201            .windows(PREFIX.len())
202            .position(|w| w == PREFIX)
203            .ok_or(LayoutError::MissingHeader("authorization"))?;
204        let value_start = prefix_at + PREFIX.len();
205        let value_end = value_start
206            + sent[value_start..]
207                .windows(2)
208                .position(|w| w == b"\r\n")
209                .ok_or(LayoutError::MissingHeader("authorization"))?;
210
211        Ok(Self::revealing(
212            [0..value_start, value_end..sent.len()],
213            sent.len(),
214        ))
215    }
216
217    /// The identity response: the two identity members with their full delimiters,
218    /// and nothing else.
219    ///
220    /// Each member is revealed whole -- delimiter, value and closing byte -- so the
221    /// verifier reads that field's value rather than a substring of a neighbouring
222    /// one, and so the match sits inside a single revealed run rather than being
223    /// spliced out of several. Everything between and around them is committed.
224    ///
225    /// # What committing the rest costs, and why it is taken
226    ///
227    /// Every reader on the verifying side scans revealed bytes: the per-range field
228    /// read and the cross-range delimiter count alike. A commitment is invisible to
229    /// all of them. So a response that genuinely names an authoritative field twice
230    /// lets a prover commit the real member and reveal the one it composed, and
231    /// both checks then see exactly one. Uniqueness is a property of the document,
232    /// and this establishes it over a part.
233    ///
234    /// Reaching that needs the PLATFORM to emit the duplicate. ASM-PROV-06 assumes
235    /// it does not, and JSON escaping keeps a `","field":"` delimiter out of any
236    /// value the account controls -- a quote inside a string is written `\"`, which
237    /// does not match the template. A duplicate that reaches the REVEALED bytes is
238    /// still caught on chain, in either range layout.
239    ///
240    /// What the commitments buy is that the rest of the response never reaches the
241    /// chain. `GET /user` under an OAuth client holding a `user`-family scope
242    /// returns the account's plan, private-repository counts, disk usage and
243    /// two-factor state; revealing the response whole would publish all of it,
244    /// permanently, for every bind.
245    ///
246    /// `id_field` and `handle_field` are both bare `&str`, and a call that
247    /// transposes them still finds both members, still reveals both, and still
248    /// tiles -- it just names the handle as the account's immutable identifier.
249    /// Nothing downstream sees the swap: the layout is well formed and the
250    /// verifier reads what it was given, so the mistake surfaces on chain as an
251    /// offset rather than as a name. `IdShape` sitting between the two is luck,
252    /// not protection. Both names come from one profile, so keep them together at
253    /// the call site.
254    ///
255    /// The arguments are still taken and still checked. A response missing either
256    /// member is a failure now rather than at the verifier, where the reason would
257    /// be an offset rather than a name.
258    pub fn identity_response(
259        recv: &[u8],
260        id_field: &str,
261        id_shape: IdShape,
262        handle_field: &str,
263    ) -> Result<Self, LayoutError> {
264        // The bare-integer form takes its structural terminator with it, which is
265        // what proves the revealed digits are the whole number.
266        let id =
267            compute_id_snippet_range(recv, id_field, id_shape == IdShape::JsonString)
268                .ok_or_else(|| LayoutError::MissingField(id_field.into()))?;
269        let handle = compute_field_snippet_range(recv, handle_field)
270            .ok_or_else(|| LayoutError::MissingField(handle_field.into()))?;
271
272        // JSON member order is not fixed; `Self::revealing` sorts, so this does
273        // not assume one.
274        Ok(Self::revealing([id, handle], recv.len()))
275    }
276}

282    /// The property every layout must have, checked directly rather than
283    /// inferred from the ranges looking plausible.
284    fn tiles(l: &Layout, len: usize) -> bool {
285        let mut spans: Vec<Range<usize>> =
286            l.reveal.iter().chain(l.commit.iter()).cloned().collect();
287        spans.sort_by_key(|r| r.start);
288        let mut at = 0usize;
289        for s in spans {
290            if s.start != at || s.end <= s.start {
291                return false;
292            }
293            at = s.end;
294        }
295        at == len
296    }

301    #[test]
302    fn a_bearer_split_by_chunk_framing_is_refused() {
303        // The session Rust actually runs. Framing inside the committed range
304        // means the circuit opens bytes the token service never returned, and
305        // the on-chain framing check passes anyway because it reads the
306        // delimiters either side of the commitment, not its contents.
307        let mut recv = b"HTTP/1.1 200 OK\r\nTransfer-Encoding: chunked\r\n\r\n".to_vec();
308        for part in [
309            r#"{"access_token":"ghu_AA"#,
310            r#"BB","token_type":"bearer"}"#,
311        ] {
312            recv.extend_from_slice(format!("{:x}\r\n", part.len()).as_bytes());
313            recv.extend_from_slice(part.as_bytes());
314            recv.extend_from_slice(b"\r\n");
315        }
316        recv.extend_from_slice(b"0\r\n\r\n");
317        assert!(Layout::token_response(&recv).is_err());
318    }

330    #[test]
331    fn a_bearer_carrying_structural_bytes_is_committed_whole() {
332        // Only `"` closes the value. A scan stopping at `:` or `,` would
333        // commit a prefix and REVEAL the rest of the bearer.
334        let recv = [
335            b"HTTP/1.1 200 OK\r\n\r\n".as_slice(),

340        assert!(tiles(&l, recv.len()));

342        // And no revealed run holds any part of it.
343        for r in &l.reveal {
344            assert!(
345                !recv[r.clone()].windows(3).any(|w| w == b"gh:"),
346                "the bearer must not appear in a revealed range"
347            );
348        }

351    #[test]
352    fn a_header_cannot_answer_for_the_body() {
353        // The old scan started at byte zero, so a response header carrying the
354        // delimiter was matched before the body's own member.
355        let recv: &[u8] = concat!(
356            "HTTP/1.1 200 OK\r\n",
357            r#"x-echo: "access_token":"decoy""#,
358            "\r\n\r\n",
359            r#"{"access_token":"real"}"#,
360        )
361        .as_bytes();
362        let l = Layout::token_response(recv).unwrap();
363        let revealed: Vec<u8> = l
364            .reveal
365            .iter()
366            .flat_map(|r| recv[r.clone()].to_vec())
367            .collect();
368        assert_eq!(revealed, br#""access_token":"""#.to_vec());
369        // The committed run is the bearer in the BODY, not the decoy.
370        let committed = l.commit.iter().find(|r| r.len() == 4).unwrap();
371        assert_eq!(&recv[committed.clone()], b"real");

478    /// The one duplicate this layout cannot defend against, recorded so the
479    /// assumption is visible on the prover side too.
480    ///
481    /// A response naming `username` twice lets the revealed range carry one
482    /// member while the other stays committed, invisible to every reader on
483    /// chain. Reaching it needs the platform to emit that document: ASM-PROV-06
484    /// assumes it does not, and JSON escaping keeps the delimiter out of any
485    /// value the account controls. The layout picks the first match and does
486    /// not detect the second -- stated here rather than left to be discovered.
487    #[test]
488    fn a_response_naming_a_member_twice_reveals_only_one() {
489        let recv: &[u8] = b"HTTP/1.1 200 OK\r\n\r\n{\"id\":\"7\",\"username\":\"victim\",\"username\":\"alice\"}";
490        let l = Layout::identity_response(recv, "id", IdShape::JsonString, "username")
491            .unwrap();
492        assert!(tiles(&l, recv.len()));
493        let revealed: usize = l
494            .reveal
495            .iter()
496            .map(|r| {
497                recv[r.clone()]
498                    .windows(11)
499                    .filter(|w| *w == b"\"username\":")
500                    .count()
501            })
502            .sum();
503        assert_eq!(revealed, 1, "the second member is committed, not revealed");
504    }
