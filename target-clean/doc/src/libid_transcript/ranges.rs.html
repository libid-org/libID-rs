1//! TLS transcript parsing and byte-range helpers for selective disclosure.
2//!
3//! All functions operate on raw transcript bytes (`sent` / `recv`) and return
4//! `Range<usize>` offsets into them. The revealed slices become Merkle leaves
5//! that on-chain verifiers check, so every helper here fails closed: a range
6//! that cannot be located contiguously in the RAW transcript (e.g. a JSON
7//! snippet split across a chunk boundary) yields `None` rather than a
8//! mis-resolved leaf.
9

17/// Find the byte range of an HTTP header value in raw TLS data.
18pub fn find_header_range(data: &[u8], name: &str) -> Option<Range<usize>> {
19    let needle = format!("\r\n{}: ", name);
20    let needle_bytes = needle.as_bytes();
21    let start = data
22        .windows(needle_bytes.len())
23        .position(|w| w.eq_ignore_ascii_case(needle_bytes))?;
24    let value_start = start.checked_add(needle_bytes.len())?;
25    let value_end = data
26        .get(value_start..)?
27        .windows(2)
28        .position(|w| w == b"\r\n")
29        .and_then(|pos| value_start.checked_add(pos))?;
30    Some(value_start..value_end)
31}
32
33/// Extract an HTTP header value from raw TLS data.
34pub fn extract_header(data: &[u8], name: &str) -> Option<String> {
35    find_header_range(data, name)
36        .map(|range| String::from_utf8_lossy(&data[range]).to_string())
37}

39/// Find the byte range of the HTTP request line in sent data.
40pub fn find_request_line_range(sent: &[u8]) -> Range<usize> {
41    let end = sent
42        .windows(2)
43        .position(|w| w == b"\r\n")
44        .unwrap_or(sent.len());
45    0..end
46}

48/// Find the byte range of the HTTP response body in received data.
49pub fn find_response_body_range(recv: &[u8]) -> Option<Range<usize>> {
50    let marker = b"\r\n\r\n";
51    recv.windows(marker.len())
52        .position(|w| w == marker)
53        .and_then(|pos| pos.checked_add(marker.len()))
54        .map(|body_start| body_start..recv.len())
55}

57/// Extract and decode the HTTP response body from received TLS data.
58///
59/// Handles both chunked and non-chunked transfer encodings.
60pub fn extract_response_body(recv: &[u8]) -> Result<Vec<u8>> {
61    let range = find_response_body_range(recv).ok_or_else(|| Error::Transcript {
62        detail: "no response body found".into(),
63    })?;
64    let raw_body = &recv[range];
65
66    if let Some(te) = extract_header(recv, "Transfer-Encoding") {
67        if te.contains("chunked") {
68            return decode_chunked_body(raw_body);
69        }
70    }
71
72    Ok(raw_body.to_vec())
73}

75/// Join a chunked body's chunks.
76///
77/// Every malformed input is an error rather than a shorter body. The reveal
78/// ranges are computed over what this returns, so a silent truncation would
79/// have the prover select ranges over bytes the server never sent -- and the
80/// notary would sign that selection without anyone noticing.
81fn decode_chunked_body(raw: &[u8]) -> Result<Vec<u8>> {
82    let bad = |detail: &str| Error::Transcript {
83        detail: format!("chunked body: {detail}"),
84    };
85
86    let mut out = Vec::new();
87    let mut rest = raw;
88    loop {
89        let (header_len, size) = match httparse::parse_chunk_size(rest) {
90            Ok(httparse::Status::Complete(v)) => v,
91            Ok(httparse::Status::Partial) => {
92                return Err(bad("ends inside a chunk header"))
93            }
94            Err(_) => return Err(bad("chunk size is not hexadecimal")),
95        };
96        if size == 0 {
97            return Ok(out);
98        }
99        let size =
100            usize::try_from(size).map_err(|_| bad("chunk larger than this machine"))?;
101        let body_end = header_len
102            .checked_add(size)
103            .ok_or_else(|| bad("chunk length overflows"))?;
104        let chunk = rest
105            .get(header_len..body_end)
106            .ok_or_else(|| bad("chunk is shorter than its declared size"))?;
107        out.extend_from_slice(chunk);
108
109        // The CRLF that closes a chunk. Its absence means the framing is not
110        // what it claims, and the next size would be read from the wrong place.
111        let after = rest
112            .get(body_end..body_end + 2)
113            .ok_or_else(|| bad("ends before a chunk terminator"))?;
114        if after != b"\r\n" {
115            return Err(bad("chunk is not terminated by CRLF"));
116        }
117        rest = &rest[body_end + 2..];
118    }
119}

121/// Find the byte range of a JSON string field value.
122pub fn find_json_field_range(body: &[u8], field: &str) -> Option<Range<usize>> {
123    let needle = format!("\"{}\"", field);
124    let pos = body
125        .windows(needle.len())
126        .position(|w| w == needle.as_bytes())?;
127    let after_key = pos.checked_add(needle.len())?;
128    let colon = body
129        .get(after_key..)?
130        .iter()
131        .position(|&b| b == b':')?
132        .checked_add(after_key)?;
133    let after_colon = colon.checked_add(1)?;
134    let open_quote = body
135        .get(after_colon..)?
136        .iter()
137        .position(|&b| b == b'"')?
138        .checked_add(after_colon)?;
139    let after_open = open_quote.checked_add(1)?;
140    let close_quote = body
141        .get(after_open..)?
142        .iter()
143        .position(|&b| b == b'"')?
144        .checked_add(after_open)?;
145    Some(after_open..close_quote)
146}

148/// Find the byte ranges that should be revealed to the notary: the request
149/// line and the Host header.
150pub fn find_notary_reveal_ranges(sent: &[u8]) -> Vec<Range<usize>> {
151    let mut ranges = Vec::new();
152
153    let req_line = find_request_line_range(sent);
154    let req_end = req_line.end.saturating_add(2).min(sent.len());
155    ranges.push(0..req_end);
156
157    if let Some(range) = find_header_range(sent, "Host") {
158        let needle = "\r\nHost: ";
159        let prefix_start = sent
160            .windows(needle.len())
161            .position(|w| w.eq_ignore_ascii_case(needle.as_bytes()));
162        if let Some(start) = prefix_start {
163            let header_end = range.end.saturating_add(2).min(sent.len());
164            ranges.push(start..header_end);
165        }
166    }
167
168    ranges
169}

171/// Find the byte ranges to commit to in the TLSNotary presentation.
172pub fn find_presentation_commit_ranges(sent: &[u8]) -> Vec<Range<usize>> {
173    find_notary_reveal_ranges(sent)
174}

176/// Compute the absolute recv-transcript byte range for a JSON field value.
177///
178/// Given the full `recv` transcript data and a JSON field name, this function:
179/// 1. Finds the HTTP response body range in `recv`
180/// 2. Decodes the body (handling chunked transfer encoding)
181/// 3. Finds the field value range in the decoded body
182/// 4. Maps it back to an absolute range in the raw `recv` data
183///
184/// Returns the absolute byte range within `recv` that contains just the
185/// field's string value (without quotes).
186pub fn compute_field_reveal_range(recv: &[u8], field_name: &str) -> Option<Range<usize>> {
187    let body_range = find_response_body_range(recv)?;
188    let raw_body = &recv[body_range.clone()];
189    let decoded_body = extract_response_body(recv).ok()?;
190
191    // Find field in decoded body to validate it exists
192    let _decoded_field_range = find_json_field_range(&decoded_body, field_name)?;
193
194    // For the actual byte range, search in the raw body (which may include
195    // chunk framing). The field bytes are the same in both representations.
196    let raw_field_range = find_json_field_range(raw_body, field_name)?;
197
198    let start = body_range.start.checked_add(raw_field_range.start)?;
199    let end = body_range.start.checked_add(raw_field_range.end)?;
200    Some(start..end)
201}

203/// The `"key":"value"` member, from the key's opening quote through the
204/// value's closing quote.
205///
206/// Unlike [`find_json_field_range`], which returns only the value bytes, this
207/// returns the whole member -- the range a reveal layout selects.
208///
209/// # The template is the reader's
210///
211/// `CeremonyFields.tryJsonString` matches the literal `"<name>":"`, so this
212/// matches the same bytes. Anything looser picks a range the reader cannot
213/// read: a body written `"login" : "octocat"` would be revealed here and then
214/// met with `FieldNotFound` on chain, which is the same refusal reported where
215/// nobody can see why. Failing here fails it where the reason is visible.
216///
217/// Uniqueness is NOT checked here, and that is deliberate. The reader refuses
218/// a delimiter matching twice in the bytes it was shown (REQ-COMMON-19A), and
219/// which bytes those are is exactly what a layout decides -- so
220/// `identity_response` reveals one member and commits the other, and the
221/// reader sees one. Refusing a second occurrence here would only stop an
222/// honest prover from building that layout; a dishonest one does not run this
223/// code at all.
224pub fn find_json_snippet_range(body: &[u8], field: &str) -> Option<Range<usize>> {
225    JsonMember::in_body(body, field).map(|member| member.member)
226}

228/// A `"field":"value"` member, and the value inside it.
229///
230/// Two ranges rather than one because a caller that reveals the delimiters and
231/// commits the value needs both boundaries, and deriving the inner one from the
232/// outer one means restating the template -- which is a second place to change
233/// the field name and one place to forget.
234#[derive(Clone, Debug, PartialEq, Eq)]
235pub struct JsonMember {
236    /// The whole member, both delimiters included.
237    pub member: Range<usize>,
238    /// The value alone, between the quotes. Empty when the value is `""`.
239    pub value: Range<usize>,
240}
241
242impl JsonMember {
243    /// The member named `field` in `body`, with offsets INTO `body`.
244    ///
245    /// Raw bytes in, raw offsets out: this scans whatever it is handed, so a
246    /// caller passing a whole HTTP response gets whichever match comes first --
247    /// a header's, if a header carries the delimiter. [`JsonMember::in_response`]
248    /// is the one that locates the body first, and is what a caller building a
249    /// reveal layout wants.
250    ///
251    /// A constructor on the type it produces: `json_member_in` restated the type
252    /// in the function name, stranded a preposition on the end of it, and left
253    /// the coordinate system -- the thing this module gets wrong most
254    /// expensively -- unsaid.
255    ///
256    /// The template it matches, and why that template is exactly the reader's,
257    /// is argued on [`find_json_snippet_range`], which is the public face of
258    /// this scan.
259    fn in_body(body: &[u8], field: &str) -> Option<Self> {
260        let needle = format!("\"{field}\":\"");
261        let start = find_first(body, needle.as_bytes())?;
262        let value = start.checked_add(needle.len())?;
263        let close = body
264            .get(value..)?
265            .iter()
266            .position(|&b| b == b'"')?
267            .checked_add(value)?;
268        Some(Self {
269            // From the opening `"` of the key through the closing `"` of the value.
270            member: start..close.checked_add(1)?,
271            value: value..close,
272        })
273    }
274
275    /// The member named `field_name` in an HTTP response, with offsets into the
276    /// RAW `recv` transcript.
277    ///
278    /// For a caller that reveals a member's delimiters and commits what sits
279    /// between them: both boundaries come from the scan that found them, so no
280    /// caller restates the template to recover one.
281    ///
282    /// The offsets are the whole difference from `in_body`, and the reason the
283    /// two are named apart rather than left to a `find_`/`compute_` prefix
284    /// nobody can decode. A reveal layout selects ranges of the TRANSCRIPT, so a
285    /// body-relative range handed to one selects bytes somewhere up in the
286    /// response headers -- a range that is well formed, signed, and pointing at
287    /// the wrong thing.
288    pub fn in_response(recv: &[u8], field_name: &str) -> Option<Self> {
289        let body_range = find_response_body_range(recv)?;
290        let raw_body = &recv[body_range.clone()];
291        let decoded_body = extract_response_body(recv).ok()?;
292
293        // Found in both: the decoded body says the member exists, the raw body says
294        // where it sits, and the two must hold the same bytes.
295        let decoded = Self::in_body(&decoded_body, field_name)?;
296        let raw = Self::in_body(raw_body, field_name)?;
297        require_contiguous(
298            raw_body.get(raw.member.clone())?,
299            decoded_body.get(decoded.member)?,
300        )?;
301
302        let at = |offset: usize| body_range.start.checked_add(offset);
303        Some(Self {
304            member: at(raw.member.start)?..at(raw.member.end)?,
305            value: at(raw.value.start)?..at(raw.value.end)?,
306        })
307    }
308}

310/// The first occurrence of `needle`, or nothing.
311fn find_first(haystack: &[u8], needle: &[u8]) -> Option<usize> {
312    haystack.windows(needle.len()).position(|w| w == needle)
313}

315/// The raw bytes are the member, and not the member with framing through it.
316///
317/// A chunked body carries `\r\n<size>\r\n` between chunks, and that framing
318/// holds no quote, comma or brace -- so a member split across a boundary is
319/// found in the decoded body AND in the raw one, and the raw range silently
320/// spans the framing. What that range selects is not the member: revealed, it
321/// puts framing inside the handle a verifier reads; committed, it puts framing
322/// inside the bearer a circuit opens against the clean value the caller was
323/// handed. Re-framing cannot repair it, because a commitment covers one
324/// contiguous run and this member is two.
325///
326/// So the session is refused here, where the reason is a decodable body rather
327/// than an unopenable commitment three components later.
328fn require_contiguous(raw: &[u8], decoded: &[u8]) -> Option<()> {
329    (raw == decoded).then_some(())
330}

332/// Find the byte range of a bare (unquoted) JSON number snippet:
333/// `"key":<number>,`. The range runs from the key's opening `"` through the
334/// trailing `,` that follows the number (matching the on-chain `idSuffix=,`).
335///
336/// Returns `None` only when neither a `,` nor a `}` terminator follows the
337/// number; both terminators are included in the range (on-chain `_extractId`
338/// scans digits and stops at either).
339pub fn find_json_bare_snippet_range(body: &[u8], field: &str) -> Option<Range<usize>> {
340    let needle = format!("\"{field}\":");
341    let start = find_first(body, needle.as_bytes())?;
342    let digits = start.checked_add(needle.len())?;
343    // Bound the number by the first `,` or `}` after the colon.
344    let term = body
345        .get(digits..)?
346        .iter()
347        .position(|&b| b == b',' || b == b'}')?
348        .checked_add(digits)?;
349    // The terminator is revealed with the digits: it is what proves they are
350    // the whole number rather than a prefix of a longer one, and
351    // `CeremonyFields.tryJsonInteger` refuses any other byte there.
352    Some(start..term.checked_add(1)?)
353}

355/// Like [`compute_field_reveal_range`] but returns the range covering the
356/// full JSON snippet `"key":"value"` instead of just the value.
357///
358/// The revealed bytes become a Merkle leaf that the contract can verify
359/// against the expected `abi.encodePacked(handlePrefix, username, '"')`.
360pub fn compute_field_snippet_range(
361    recv: &[u8],
362    field_name: &str,
363) -> Option<Range<usize>> {
364    JsonMember::in_response(recv, field_name).map(|found| found.member)
365}

367/// Like [`compute_id_snippet_range`] but only matches `field_name` after the
368/// first occurrence of `anchor_field` (disambiguates a non-unique id field).
369pub fn compute_id_snippet_range_after(
370    recv: &[u8],
371    field_name: &str,
372    quoted: bool,
373    anchor_field: &str,
374) -> Option<Range<usize>> {
375    let body_range = find_response_body_range(recv)?;
376    let raw_body = &recv[body_range.clone()];
377
378    // Include the `:` so we match the JSON KEY `"user":` — not a substring of a
379    // user-controlled body field (whose quotes are JSON-escaped) nor a sibling
380    // key like `"user_view_type"`.
381    let anchor_needle = format!("\"{}\":", anchor_field);
382
383    // Validate the anchored id against the DECODED body (chunk-framing stripped),
384    // so a body that is chunked or contains decoy bytes can't drive the result.
385    // The bytes it finds are kept, to be compared with the raw ones below.
386    let decoded = extract_response_body(recv).ok()?;
387    let decoded_member = {
388        let danchor = decoded
389            .windows(anchor_needle.len())
390            .position(|w| w == anchor_needle.as_bytes())?;
391        let from = danchor.checked_add(anchor_needle.len())?;
392        let dsub = decoded.get(from..)?;
393        let rel = if quoted {
394            find_json_snippet_range(dsub, field_name)?
395        } else {
396            find_json_bare_snippet_range(dsub, field_name)?
397        };
398        dsub.get(rel)?
399    };
400
401    // The Merkle leaf is over the RAW transcript, so map the range there.
402    let anchor_pos = raw_body
403        .windows(anchor_needle.len())
404        .position(|w| w == anchor_needle.as_bytes())?;
405    let search_from = anchor_pos.checked_add(anchor_needle.len())?;
406    let sub = raw_body.get(search_from..)?;
407
408    let rel = if quoted {
409        find_json_snippet_range(sub, field_name)?
410    } else {
411        find_json_bare_snippet_range(sub, field_name)?
412    };
413    require_contiguous(sub.get(rel.clone())?, decoded_member)?;
414
415    let base = body_range.start.checked_add(search_from)?;
416    Some(base.checked_add(rel.start)?..base.checked_add(rel.end)?)
417}

419/// Compute the absolute recv-transcript range for an id snippet, dispatching on
420/// quotedness: `quoted` → `"id":"<id>"`, otherwise the bare `"id":<n>[,}]` form.
421///
422/// Returns `None` if the field is absent. Both `,`- and `}`-terminated bare
423/// numbers are matched (on-chain `_extractId` scans digits past either).
424pub fn compute_id_snippet_range(
425    recv: &[u8],
426    field_name: &str,
427    quoted: bool,
428) -> Option<Range<usize>> {
429    if quoted {
430        return compute_field_snippet_range(recv, field_name);
431    }
432    let body_range = find_response_body_range(recv)?;
433    let raw_body = &recv[body_range.clone()];
434    let decoded_body = extract_response_body(recv).ok()?;
435
436    let decoded_range = find_json_bare_snippet_range(&decoded_body, field_name)?;
437    let raw_snippet_range = find_json_bare_snippet_range(raw_body, field_name)?;
438    require_contiguous(
439        raw_body.get(raw_snippet_range.clone())?,
440        decoded_body.get(decoded_range)?,
441    )?;
442
443    let start = body_range.start.checked_add(raw_snippet_range.start)?;
444    let end = body_range.start.checked_add(raw_snippet_range.end)?;
445    Some(start..end)
446}

602    /// A chunked response whose `field` value is cut in half by a chunk
603    /// boundary. The framing carries no quote, comma or brace, so every scan
604    /// here runs straight through it.
605    fn straddling(head: &str, tail: &str) -> Vec<u8> {
606        let mut out = b"HTTP/1.1 200 OK\r\nTransfer-Encoding: chunked\r\n\r\n".to_vec();
607        for part in [head, tail] {
608            out.extend_from_slice(format!("{:x}\r\n", part.len()).as_bytes());
609            out.extend_from_slice(part.as_bytes());
610            out.extend_from_slice(b"\r\n");
611        }
612        out.extend_from_slice(b"0\r\n\r\n");
613        out
614    }

616    /// The property every caller of `JsonMember::in_response` depends on: the
617    /// value sits inside the member, and what the member holds either side of
618    /// it is exactly the two delimiters. A boundary that drifts breaks this
619    /// before it reaches a layout, where the symptom is a committed bearer with
620    /// a quote in it.
621    fn assert_brackets(recv: &[u8], found: &JsonMember, field: &str, value: &[u8]) {
622        assert!(
623            found.member.start <= found.value.start
624                && found.value.end <= found.member.end,
625            "the value must sit inside the member"
626        );
627        assert_eq!(&recv[found.value.clone()], value, "value bytes");
628        assert_eq!(
629            &recv[found.member.start..found.value.start],
630            format!("\"{field}\":\"").as_bytes(),
631            "opening delimiter"
632        );
633        assert_eq!(
634            &recv[found.value.end..found.member.end],
635            b"\"",
636            "closing quote"
637        );
638    }
